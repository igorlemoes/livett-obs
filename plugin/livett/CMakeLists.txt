cmake_minimum_required(VERSION 3.24)
project(livett VERSION 0.2.0 LANGUAGES C CXX)

# Build against the OBS 32.2.2 sources + obs-deps Qt6 fetched under ../build-deps (see ../README.md).
set(OBS_SRC "${CMAKE_CURRENT_LIST_DIR}/../build-deps/obs-studio" CACHE PATH "OBS Studio sources (tag 32.2.2)")
set(OBS_QT_PREFIX "${CMAKE_CURRENT_LIST_DIR}/../build-deps/qt6/x" CACHE PATH "obs-deps Qt6 prefix")
set(OBS_IMPORT_LIBS "${CMAKE_CURRENT_LIST_DIR}/../build/importlibs" CACHE PATH "obs.lib / obs-frontend-api.lib (plugin/tools/gen-importlibs.ps1)")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
list(APPEND CMAKE_PREFIX_PATH "${OBS_QT_PREFIX}")

find_package(Qt6 REQUIRED COMPONENTS Core Widgets Network)

# libobs headers include "obsconfig.h", generated by the OBS build; the plugin needs none of its settings.
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated")
file(WRITE "${CMAKE_BINARY_DIR}/generated/obsconfig.h" "#pragma once\n#define OBS_RELEASE_CANDIDATE 0\n#define OBS_BETA 0\n")

add_library(livett MODULE src/plugin-main.cpp src/obs-api.cpp src/obs-service.cpp src/phone-camera.cpp)
target_include_directories(livett PRIVATE
  "${OBS_SRC}/libobs" "${OBS_SRC}/frontend/api" "${OBS_SRC}/plugins/obs-browser/panel" "${CMAKE_BINARY_DIR}/generated")
target_compile_definitions(livett PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(livett PRIVATE Qt6::Core Qt6::Widgets Qt6::Network
  "${OBS_IMPORT_LIBS}/obs.lib" "${OBS_IMPORT_LIBS}/obs-frontend-api.lib")
if(MSVC)
  target_compile_options(livett PRIVATE /W3 /utf-8 /permissive-)
endif()
set_target_properties(livett PROPERTIES PREFIX "")
